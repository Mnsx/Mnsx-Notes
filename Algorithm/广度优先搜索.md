> **广度优先搜索**，可以找出两样东西之间的最短距离

## 图

图由节点和边组成，一个节点可能与众多节点直接相连，这些节点被称为邻居

## 简单实现

```c++
void bfsTemplate(int start, vector<vector<int>> & graph) {

    int n = graph.size();

    queue<int> q;
    vector<bool> visited(n, false);

    q.push(start);
    visited[start] = true;

    while (!q.empty()) {
        int node = q.front();
        q.pop();

        cout << node << " ";

        for (int neighbor : graph[node]) {

            if (!visited[neighbor]) {

                visited[neighbor] = true;
                q.push(neighbor);
            }
        }
    }
}
```
