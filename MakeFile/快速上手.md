# 快速上手

```makefile
# 指定CMAKE的最低执行版本
cmake_minimum_required(VERSION 4.1)

# 设置项目名称
project(Test)

# 设置C++编译器版本
set(CMAKE_CXX_STANDARD 20)

# 自动创建项目文件
file(MAKE_DIRECTORY ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/include ${CMAKE_SOURCE_DIR}/bin ${CMAKE_SOURCE_DIR}/test)

# 设置二进制可执行文件的生成位置
set(EXECUTABLE_OUTPUT_PATH ${PROJECT_SOURCE_DIR}/bin)

# 指定include文件
include_directories(include)

# 将文件添加到可执行文件中，并且设置可执行文件名称
add_executable(Test main.cpp)
```
